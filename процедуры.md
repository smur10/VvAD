CREATE PROCEDURE sp\_get\_unpaid\_contracts()

BEGIN

&#x20;   -- Реализация запроса №1 из раздела 4

&#x20;   WITH unpaid\_contracts AS (

&#x20;       SELECT 

&#x20;           l.language\_name,

&#x20;           COUNT(\*) AS unpaid\_count,

&#x20;           SUM(co.price) AS total\_debt

&#x20;       FROM contracts c

&#x20;       JOIN courses co ON co.id\_course = c.id\_course

&#x20;       JOIN languages l ON l.id\_language = co.id\_language

&#x20;       WHERE c.payment\_status = 'не\_оплачен'

&#x20;       GROUP BY l.language\_name

&#x20;   )

&#x20;   SELECT 

&#x20;       language\_name,

&#x20;       unpaid\_count,

&#x20;       total\_debt,

&#x20;       ROUND(100.0 \* unpaid\_count / (SELECT COUNT(\*) FROM contracts), 2) AS percent\_of\_all\_contracts

&#x20;   FROM unpaid\_contracts

&#x20;   ORDER BY total\_debt DESC, unpaid\_count DESC;

END //

CREATE PROCEDURE sp\_get\_free\_places(

&#x20;   IN p\_language\_name VARCHAR(50),

&#x20;   IN p\_level\_name    VARCHAR(50)

)

BEGIN

&#x20;   DECLARE v\_language\_count INT DEFAULT 0;

&#x20;   DECLARE v\_level\_count INT DEFAULT 0;

&#x20;   DECLARE v\_msg VARCHAR(255); -- 1. Добавляем переменную для сообщения



&#x20;   -- === Обработка ошибок: проверка входных параметров ===

&#x20;   IF p\_language\_name IS NULL OR p\_language\_name = '' THEN

&#x20;       SIGNAL SQLSTATE '45000'

&#x20;       SET MESSAGE\_TEXT = 'Ошибка: Не указан язык';

&#x20;   END IF;



&#x20;   IF p\_level\_name IS NULL OR p\_level\_name = '' THEN

&#x20;       SIGNAL SQLSTATE '45000'

&#x20;       SET MESSAGE\_TEXT = 'Ошибка: Не указан уровень';

&#x20;   END IF;



&#x20;   -- Проверка существования языка

&#x20;   SELECT COUNT(\*) INTO v\_language\_count 

&#x20;   FROM languages WHERE language\_name = p\_language\_name;

&#x20;   

&#x20;   IF v\_language\_count = 0 THEN

&#x20;       -- 2. Сначала формируем текст, потом кидаем сигнал

&#x20;       SET v\_msg = CONCAT('Ошибка: Язык "', p\_language\_name, '" не найден');

&#x20;       SIGNAL SQLSTATE '45000' SET MESSAGE\_TEXT = v\_msg;

&#x20;   END IF;



&#x20;   -- Проверка существования уровня

&#x20;   SELECT COUNT(\*) INTO v\_level\_count 

&#x20;   FROM levels WHERE level\_name = p\_level\_name;

&#x20;   

&#x20;   IF v\_level\_count = 0 THEN

&#x20;       -- 2. Аналогично для уровня

&#x20;       SET v\_msg = CONCAT('Ошибка: Уровень "', p\_level\_name, '" не найден');

&#x20;       SIGNAL SQLSTATE '45000' SET MESSAGE\_TEXT = v\_msg;

&#x20;   END IF;



&#x20;   -- === Основной запрос ===

&#x20;   SELECT 

&#x20;       l.language\_name,

&#x20;       lv.level\_name,

&#x20;       g.max\_students,

&#x20;       COUNT(e.id\_enrollment) AS current\_students,

&#x20;       (g.max\_students - COUNT(e.id\_enrollment)) AS free\_places,

&#x20;       ROUND(100.0 \* COUNT(e.id\_enrollment) / g.max\_students, 2) AS occupancy\_percent

&#x20;   FROM study\_groups g

&#x20;   JOIN courses c ON c.id\_course = g.id\_course

&#x20;   JOIN languages l ON l.id\_language = c.id\_language

&#x20;   JOIN levels lv ON lv.id\_level = c.id\_level

&#x20;   LEFT JOIN enrollments e ON e.id\_group = g.id\_group 

&#x20;       AND e.enrollment\_status = 'активен'

&#x20;   WHERE l.language\_name = p\_language\_name

&#x20;     AND lv.level\_name = p\_level\_name

&#x20;   GROUP BY g.id\_group, l.language\_name, lv.level\_name, g.max\_students

&#x20;   ORDER BY free\_places DESC;



END //

CREATE PROCEDURE sp\_get\_successful\_students()

BEGIN

&#x20;   -- Реализация запроса №5 из раздела 4

&#x20;   WITH successful\_students AS (

&#x20;       SELECT 

&#x20;           CONCAT(s.surname, ' ', s.first\_name) AS student\_name,

&#x20;           l.language\_name,

&#x20;           lv.level\_name,

&#x20;           COUNT(g.id\_grade) AS exams\_passed,

&#x20;           ROUND(AVG((g.grammar + g.phonetics + g.speaking) / 3.0), 2) AS average\_grade,

&#x20;           MAX(e.exam\_date) AS last\_exam\_date

&#x20;       FROM grades g

&#x20;       JOIN exams e ON e.id\_exam = g.id\_exam

&#x20;       JOIN enrollments en ON en.id\_student = g.id\_student 

&#x20;           AND en.id\_group = e.id\_group

&#x20;       JOIN study\_groups sg ON sg.id\_group = e.id\_group

&#x20;       JOIN courses c ON c.id\_course = sg.id\_course

&#x20;       JOIN languages l ON l.id\_language = c.id\_language

&#x20;       JOIN levels lv ON lv.id\_level = c.id\_level

&#x20;       JOIN students s ON s.id\_student = g.id\_student

&#x20;       WHERE g.grammar >= 3 

&#x20;         AND g.phonetics >= 3 

&#x20;         AND g.speaking >= 3

&#x20;       GROUP BY s.id\_student, s.surname, s.first\_name, l.language\_name, lv.level\_name

&#x20;       HAVING AVG((g.grammar + g.phonetics + g.speaking) / 3.0) >= 4.0

&#x20;   )

&#x20;   SELECT 

&#x20;       student\_name,

&#x20;       language\_name,

&#x20;       level\_name,

&#x20;       exams\_passed,

&#x20;       average\_grade,

&#x20;       last\_exam\_date

&#x20;   FROM successful\_students

&#x20;   ORDER BY average\_grade DESC, exams\_passed DESC, student\_name;

END //

CREATE PROCEDURE sp\_get\_teacher\_efficiency()

BEGIN

&#x20;   -- Реализация Сложного запроса №1 из раздела 4

&#x20;   WITH teacher\_stats AS (

&#x20;       SELECT 

&#x20;           t.id\_teacher,

&#x20;           CONCAT(t.surname, ' ', t.first\_name) AS teacher\_name,

&#x20;           COUNT(DISTINCT sg.id\_group) AS groups\_count,

&#x20;           SUM(sg.max\_students) AS total\_max\_students,

&#x20;           COUNT(DISTINCT e.id\_enrollment) AS total\_students,

&#x20;           t.salary,

&#x20;           ROUND(AVG(sg.max\_students), 2) AS avg\_group\_size,

&#x20;           ROUND(100.0 \* COUNT(DISTINCT e.id\_enrollment) / SUM(sg.max\_students), 2) AS avg\_occupancy\_percent,

&#x20;           SUM(co.price) AS total\_revenue

&#x20;       FROM teachers t

&#x20;       JOIN study\_groups sg ON sg.id\_teacher = t.id\_teacher

&#x20;       LEFT JOIN enrollments e ON e.id\_group = sg.id\_group 

&#x20;           AND e.enrollment\_status = 'активен'

&#x20;       JOIN courses co ON co.id\_course = sg.id\_course

&#x20;       GROUP BY t.id\_teacher, t.surname, t.first\_name, t.salary

&#x20;       HAVING COUNT(DISTINCT sg.id\_group) > 1

&#x20;   ),

&#x20;   overall\_avg AS (

&#x20;       SELECT 

&#x20;           ROUND(AVG(groups\_count), 2) AS avg\_groups,

&#x20;           ROUND(AVG(avg\_occupancy\_percent), 2) AS avg\_occupancy,

&#x20;           ROUND(AVG(total\_revenue), 2) AS avg\_revenue

&#x20;       FROM teacher\_stats

&#x20;   )

&#x20;   SELECT 

&#x20;       ts.teacher\_name,

&#x20;       ts.groups\_count,

&#x20;       ts.total\_students,

&#x20;       ts.avg\_group\_size,

&#x20;       ts.avg\_occupancy\_percent,

&#x20;       ts.total\_revenue,

&#x20;       ts.salary,

&#x20;       ROUND(ts.salary / ts.groups\_count, 2) AS salary\_per\_group,

&#x20;       ROUND(ts.avg\_occupancy\_percent - oa.avg\_occupancy, 2) AS occupancy\_diff,

&#x20;       ROUND(ts.total\_revenue - oa.avg\_revenue, 2) AS revenue\_diff

&#x20;   FROM teacher\_stats ts

&#x20;   CROSS JOIN overall\_avg oa

&#x20;   ORDER BY ts.total\_revenue DESC, ts.avg\_occupancy\_percent DESC;

END //

CREATE PROCEDURE sp\_get\_student\_performance\_comparison()

BEGIN

&#x20;   -- Реализация обновлённого Сложного запроса №3 из раздела 4

&#x20;   WITH student\_avg AS (

&#x20;       SELECT 

&#x20;           s.id\_student,

&#x20;           CONCAT(s.surname, ' ', s.first\_name) AS student\_name,

&#x20;           l.language\_name,

&#x20;           lv.level\_name,

&#x20;           ROUND(AVG((g.grammar + g.phonetics + g.speaking) / 3.0), 2) AS student\_average,

&#x20;           COUNT(g.id\_grade) AS exams\_passed,

&#x20;           MAX(e.exam\_date) AS last\_exam\_date

&#x20;       FROM grades g

&#x20;       JOIN exams e ON e.id\_exam = g.id\_exam

&#x20;       JOIN enrollments en ON en.id\_student = g.id\_student 

&#x20;           AND en.id\_group = e.id\_group

&#x20;       JOIN study\_groups sg ON sg.id\_group = e.id\_group

&#x20;       JOIN courses c ON c.id\_course = sg.id\_course

&#x20;       JOIN languages l ON l.id\_language = c.id\_language

&#x20;       JOIN levels lv ON lv.id\_level = c.id\_level

&#x20;       JOIN students s ON s.id\_student = g.id\_student

&#x20;       WHERE g.grammar >= 3 

&#x20;         AND g.phonetics >= 3 

&#x20;         AND g.speaking >= 3

&#x20;       GROUP BY s.id\_student, s.surname, s.first\_name, l.language\_name, lv.level\_name

&#x20;       HAVING AVG((g.grammar + g.phonetics + g.speaking) / 3.0) >= 4.0

&#x20;   ),

&#x20;   group\_avg AS (

&#x20;       SELECT 

&#x20;           l.language\_name,

&#x20;           lv.level\_name,

&#x20;           ROUND(AVG((g.grammar + g.phonetics + g.speaking) / 3.0), 2) AS group\_average

&#x20;       FROM grades g

&#x20;       JOIN exams e ON e.id\_exam = g.id\_exam

&#x20;       JOIN enrollments en ON en.id\_student = g.id\_student 

&#x20;           AND en.id\_group = e.id\_group

&#x20;       JOIN study\_groups sg ON sg.id\_group = e.id\_group

&#x20;       JOIN courses c ON c.id\_course = sg.id\_course

&#x20;       JOIN languages l ON l.id\_language = c.id\_language

&#x20;       JOIN levels lv ON lv.id\_level = c.id\_level

&#x20;       WHERE g.grammar >= 3 

&#x20;         AND g.phonetics >= 3 

&#x20;         AND g.speaking >= 3

&#x20;       GROUP BY l.language\_name, lv.level\_name

&#x20;   )

&#x20;   SELECT 

&#x20;       sa.student\_name,

&#x20;       sa.language\_name,

&#x20;       sa.level\_name,

&#x20;       sa.exams\_passed,

&#x20;       sa.student\_average,

&#x20;       ga.group\_average,

&#x20;       ROUND(sa.student\_average - ga.group\_average, 2) AS difference\_from\_average,

&#x20;       CASE 

&#x20;           WHEN sa.student\_average - ga.group\_average > 0.5 THEN 'Значительно выше среднего'

&#x20;           WHEN sa.student\_average - ga.group\_average > 0.0 THEN 'Выше среднего'

&#x20;           WHEN sa.student\_average - ga.group\_average < -0.1 AND sa.student\_average - ga.group\_average > -0.5 THEN 'Немного ниже среднего'

&#x20;           WHEN sa.student\_average - ga.group\_average < -0.5 THEN 'Значительно ниже среднего'

&#x20;           ELSE 'На уровне среднего'

&#x20;       END AS performance\_level,

&#x20;       sa.last\_exam\_date

&#x20;   FROM student\_avg sa

&#x20;   JOIN group\_avg ga 

&#x20;       ON ga.language\_name = sa.language\_name 

&#x20;      AND ga.level\_name = sa.level\_name

&#x20;   ORDER BY difference\_from\_average DESC, sa.student\_average DESC, sa.student\_name;

END //

CREATE PROCEDURE sp\_get\_top\_courses(

&#x20;   IN p\_date\_from DATE,

&#x20;   IN p\_date\_to   DATE

)

BEGIN

&#x20;   DECLARE v\_start\_date DATE;

&#x20;   DECLARE v\_end\_date   DATE;



&#x20;   -- Обработка параметров дат (если не указаны — берём последние 12 месяцев)

&#x20;   SET v\_start\_date = IFNULL(p\_date\_from, DATE\_SUB(CURDATE(), INTERVAL 12 MONTH));

&#x20;   SET v\_end\_date   = IFNULL(p\_date\_to, CURDATE());



&#x20;   -- Проверка корректности периода

&#x20;   IF v\_start\_date > v\_end\_date THEN

&#x20;       SIGNAL SQLSTATE '45000'

&#x20;       SET MESSAGE\_TEXT = 'Ошибка: Начальная дата не может быть позже конечной';

&#x20;   END IF;



&#x20;   -- Реализация Сложного запроса №4 из раздела 4

&#x20;   WITH course\_stats AS (

&#x20;       SELECT 

&#x20;           l.language\_name,

&#x20;           lv.level\_name,

&#x20;           COUNT(c.id\_contract) AS contracts\_count,

&#x20;           SUM(co.price) AS total\_revenue,

&#x20;           COUNT(DISTINCT sg.id\_group) AS groups\_count,

&#x20;           ROUND(AVG(sg.max\_students), 2) AS avg\_group\_size,

&#x20;           COUNT(DISTINCT e.id\_student) AS unique\_students

&#x20;       FROM courses co

&#x20;       JOIN languages l ON l.id\_language = co.id\_language

&#x20;       JOIN levels lv ON lv.id\_level = co.id\_level

&#x20;       LEFT JOIN contracts c ON c.id\_course = co.id\_course

&#x20;           AND c.contract\_date BETWEEN v\_start\_date AND v\_end\_date

&#x20;       LEFT JOIN study\_groups sg ON sg.id\_course = co.id\_course

&#x20;       LEFT JOIN enrollments e ON e.id\_group = sg.id\_group 

&#x20;           AND e.enrollment\_status = 'активен'

&#x20;       GROUP BY l.language\_name, lv.level\_name, co.id\_course

&#x20;   )

&#x20;   SELECT 

&#x20;       language\_name,

&#x20;       level\_name,

&#x20;       contracts\_count,

&#x20;       total\_revenue,

&#x20;       groups\_count,

&#x20;       avg\_group\_size,

&#x20;       unique\_students,

&#x20;       ROUND(100.0 \* contracts\_count / (SELECT COUNT(\*) FROM contracts 

&#x20;                                        WHERE contract\_date BETWEEN v\_start\_date AND v\_end\_date), 2) 

&#x20;           AS percent\_of\_all\_contracts,

&#x20;       ROUND(total\_revenue / NULLIF(contracts\_count, 0), 2) AS revenue\_per\_contract,

&#x20;       RANK() OVER (ORDER BY total\_revenue DESC, contracts\_count DESC) AS popularity\_rank

&#x20;   FROM course\_stats

&#x20;   ORDER BY popularity\_rank

&#x20;   LIMIT 5;
